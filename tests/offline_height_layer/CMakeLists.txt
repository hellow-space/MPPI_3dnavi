cmake_minimum_required(VERSION 3.16)
project(offline_height_layer LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(GTest REQUIRED)
set(dependencies grid_map_core grid_map_ros grid_map_msgs nav2_costmap_2d nav2_util
  pluginlib rclcpp rclcpp_lifecycle tf2_ros)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
set(layer_root "${CMAKE_CURRENT_SOURCE_DIR}/../../src/height_obstacle_layer")
add_executable(test_height_layer "${layer_root}/test/test_layer.cpp"
  "${layer_root}/src/height_obstacle_layer.cpp")
target_compile_features(test_height_layer PRIVATE cxx_std_17)
target_compile_options(test_height_layer PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
target_include_directories(test_height_layer PRIVATE "${layer_root}/include")
target_link_libraries(test_height_layer GTest::gtest_main)
ament_target_dependencies(test_height_layer ${dependencies})
enable_testing()
add_test(NAME height_layer_integration COMMAND test_height_layer)

add_executable(test_sampling "${layer_root}/test/test_sampling.cpp")
target_compile_features(test_sampling PRIVATE cxx_std_17)
target_compile_options(test_sampling PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
target_include_directories(test_sampling PRIVATE "${layer_root}/include")
target_link_libraries(test_sampling GTest::gtest_main)
ament_target_dependencies(test_sampling grid_map_core)
add_test(NAME neighborhood_sampling COMMAND test_sampling)

add_executable(benchmark_scan benchmark_scan.cpp)
target_compile_features(benchmark_scan PRIVATE cxx_std_17)
target_compile_options(benchmark_scan PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
target_include_directories(benchmark_scan PRIVATE "${layer_root}/include")
ament_target_dependencies(benchmark_scan grid_map_core)
