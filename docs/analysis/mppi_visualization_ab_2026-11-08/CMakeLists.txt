cmake_minimum_required(VERSION 3.16)
project(mppi_ab_timing LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
add_library(mppi_ab_timing SHARED timing.cpp)
target_compile_features(mppi_ab_timing PRIVATE cxx_std_17)
target_compile_definitions(mppi_ab_timing PRIVATE XTENSOR_USE_XSIMD XTENSOR_DEFAULT_ALIGNMENT=32)
ament_target_dependencies(mppi_ab_timing nav2_mppi_controller)
target_link_libraries(mppi_ab_timing dl)
